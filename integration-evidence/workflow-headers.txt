on:
  workflow_dispatch:
    inputs:
      as_of:
        description: "UTC ISO-8601 timestamp; leave empty to use the current time"
        required: false
        default: ""
        type: string
      seed:
        description: "Optional deterministic random-selection seed"
        required: false
        default: ""
        type: string
      trade_id:
        description: "Optional exact eligible trade ID instead of random selection"
        required: false
        default: ""
        type: string
permissions:
  actions: read
  contents: read

concurrency:
  group: filing-simulation
  cancel-in-progress: false

jobs:
  simulate:
    if: ${{ github.run_attempt == 1 && github.ref == format('refs/heads/{0}', github.event.repository.default_branch) }}
    runs-on: ubuntu-latest
    timeout-minutes: 20

    steps:
      - name: Resolve immutable simulation cutoff
        id: parameters
        shell: bash
        env:
          AS_OF_INPUT: ${{ inputs.as_of }}
          SEED_INPUT: ${{ inputs.seed }}
          TRADE_ID_INPUT: ${{ inputs.trade_id }}
        run: |
          set -euo pipefail
          if [[ -n "$SEED_INPUT" && -n "$TRADE_ID_INPUT" ]]; then
            echo "Provide either seed or trade_id, not both." >&2
            exit 1
          fi
          as_of="$AS_OF_INPUT"
          if [[ -z "$as_of" ]]; then
            as_of="$(date -u +%Y-%m-%dT%H:%M:%SZ)"
          fi
          if [[ "$as_of" == *$'\r'* || "$as_of" == *$'\n'* ]]; then
            echo "as_of must be one ISO-8601 timestamp on a single line." >&2
            exit 1
          fi
          echo "as_of=$as_of" >> "$GITHUB_OUTPUT"

      - name: Check out repository
        uses: actions/checkout@v6

      - name: Set up Python
        uses: actions/setup-python@v7
        with:
          python-version: "3.12"

      - name: Install test dependency
        shell: bash
        run: python -m pip install pytest==9.0.2

      - name: Run simulator tests
        shell: bash
        run: python -m pytest -q tests/test_filing_simulation.py

      - name: Restore immutable production inputs

          [[ "${actual_number}" == "${PROJECT_NUMBER}" ]] || { echo "Immutable project boundary mismatch." >&2; exit 1; }

          if gcloud storage ls "${marker_uri}" >/dev/null 2>&1; then
            echo "Phase 3 admin failure diagnostic marker already exists; skipping duplicate execution."
            exit 0
          fi
          printf 'claimed_at=%s\nworkflow_run=%s\nsource_revision=%s\n' "$(date -u +%Y-%m-%dT%H:%M:%SZ)" "${GITHUB_RUN_ID}" "${GITHUB_SHA}" > oneshot-marker.txt
          gcloud storage cp oneshot-marker.txt "${marker_uri}" >/dev/null

          gcloud run jobs describe "${ADMIN_JOB}" --project "${PROJECT_ID}" --region "${REGION}" --format=json > admin-current.json
          container_json="$(jq -c 'first(.. | objects | select(has("containers")) | .containers[0])' admin-current.json)"
          command_json="$(jq -c '.command // []' <<<"${container_json}")"
          args_json="$(jq -c '.args // []' <<<"${container_json}")"
          [[ "${command_json}" == '["python"]' ]] || { echo "Unexpected admin command baseline." >&2; exit 1; }
          if [[ "${args_json}" == '["-m","runtime_v2","status"]' ]]; then
            gcloud run jobs update "${ADMIN_JOB}" --project "${PROJECT_ID}" --region "${REGION}" --args=-m,runtime_v2,init-db,--with-vault --quiet >/dev/null
            args_json='["-m","runtime_v2","init-db","--with-vault"]'
          fi
          [[ "${args_json}" == '["-m","runtime_v2","init-db","--with-vault"]' ]] || { echo "Admin job is not at the Terraform Phase 3 baseline." >&2; exit 1; }

          gcloud run jobs remove-iam-policy-binding "${ADMIN_JOB}" --project "${PROJECT_ID}" --region "${REGION}" --member "${member}" --role roles/run.invoker --quiet >/dev/null 2>&1 || true
